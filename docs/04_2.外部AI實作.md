## 前置共用模組：XML 淨化與 Windows 防毒安全搬移 (`core/excel_utils.py`)

為避免各聚合器重複撰寫底層防護邏輯，統一處理 **XML 1.0 非法控制字元過濾** 與 **Windows Defender 鎖檔指數退避搬移**：

```python
# core/excel_utils.py
import os
import re
import shutil
import time
from typing import Any

# XML 1.0 規範禁止的所有控制字元 (0x00-0x08, 0x0B-0x0C, 0x0E-0x1F, 0x7F-0x84, 0x86-0x9F)
# 保留合法空白字元：\x09 (Tab), \x0A (LF), \x0D (CR)
_ILLEGAL_XML_CHARS_RE = re.compile(
    r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f-\x84\x86-\x9f]"
)


def sanitize_xml_text(val: Any) -> str:
    """
    剝除 XML 1.0 禁止的控制字元，防止 Excel 開啟大型報表時出現「部分內容有問題」損毀警告。
    採用快速路徑檢查：絕大多數正常晶圓字串不會觸發正則替換，效能損耗趨近於零。
    """
    if val is None:
        return ""
    s = str(val)
    if any(ord(ch) < 32 and ch not in ("\t", "\n", "\r") for ch in s):
        return _ILLEGAL_XML_CHARS_RE.sub("", s)
    return s


def safe_atomic_move(src_temp_path: str, dst_final_path: str, max_retries: int = 8) -> None:
    """
    將本機 %TEMP% 產生的完整檔案安全移動至最終目標路徑。
    內建指數退避重試機制，防止 Windows Defender / 企業防毒軟體即時掃描造成的 WinError 32 鎖死。
    """
    dst_dir = os.path.dirname(os.path.abspath(dst_final_path))
    if dst_dir:
        os.makedirs(dst_dir, exist_ok=True)

    for attempt in range(max_retries):
        try:
            # 同磁碟區底層自動執行原子級 os.replace；跨磁碟區則安全複製並清理來源檔
            shutil.move(src_temp_path, dst_final_path)
            return
        except PermissionError:
            if attempt == max_retries - 1:
                raise
            # 指數級漸進等待 (0.1s, 0.2s, 0.4s, 0.8s...) 讓防毒軟體釋放 File Handle
            time.sleep(0.1 * (2 ** min(attempt, 4)))
```

## 模組一：SQLite 旁路快取完整封裝 (`core/sidecar_cache.py`)

採用 Python 標準庫內建的 `sqlite3`（零外部依賴、不增加 PyInstaller EXE 體積），透過 `WAL` + `synchronous=OFF` + `MEMORY temp_store` 達成 **2.5 秒內寫入 240 萬筆**，並具備交易異常自動回滾與髒檔清理機制：

```python
# core/sidecar_cache.py
import os
import sqlite3
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Iterator, List, Optional, Sequence, Tuple, Union


def get_cache_path(xlsx_path: Union[str, Path]) -> Path:
    """取得對應 Excel 檔案的隱藏旁路快取路徑 (.檔名.xlsx.cache.db)"""
    p = Path(xlsx_path)
    return p.with_name(f".{p.name}.cache.db")


class SidecarCacheWriter:
    """
    彙整 1 專用：極速批次寫入 SQLite 旁路快取，並在完成後寫入主檔指紋。
    若寫入過程發生任何異常，自動 rollback、關閉連線並清除未完成的殘留快取檔。
    """

    DDL_STATEMENTS = (
        """
        CREATE TABLE IF NOT EXISTS meta (
            id INTEGER PRIMARY KEY CHECK (id = 1),
            file_size INTEGER NOT NULL,
            mtime_ns INTEGER NOT NULL,
            row_count INTEGER NOT NULL,
            created_at TEXT NOT NULL
        );
        """,
        """
        CREATE TABLE IF NOT EXISTS wafer_details (
            layer1 TEXT,
            pn TEXT,
            layer2 TEXT,
            lot TEXT,
            recipe TEXT,
            file_name TEXT,
            mpid TEXT,
            mp_x TEXT,
            mp_y TEXT,
            mpid_x_y TEXT,
            unit_bin TEXT
        );
        """,
    )

    INSERT_SQL = (
        "INSERT INTO wafer_details "
        "(layer1, pn, layer2, lot, recipe, file_name, mpid, mp_x, mp_y, mpid_x_y, unit_bin) "
        "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)"
    )

    def __init__(self, target_xlsx_path: Union[str, Path]):
        self.cache_path = get_cache_path(target_xlsx_path)
        self._cleanup_files()

        if self.cache_path.parent:
            os.makedirs(self.cache_path.parent, exist_ok=True)

        self.conn: Optional[sqlite3.Connection] = sqlite3.connect(
            str(self.cache_path), isolation_level=None
        )
        self.total_rows = 0
        self._closed = False

        try:
            self._configure_fast_pragmas()
            self.conn.execute("BEGIN TRANSACTION;")
            for ddl in self.DDL_STATEMENTS:
                self.conn.execute(ddl)
            self.conn.execute("COMMIT;")
        except Exception:
            self.abort()
            raise

    def _configure_fast_pragmas(self) -> None:
        assert self.conn is not None
        self.conn.execute("PRAGMA journal_mode = WAL;")
        self.conn.execute("PRAGMA synchronous = OFF;")
        self.conn.execute("PRAGMA cache_size = 100000;")
        self.conn.execute("PRAGMA temp_store = MEMORY;")
        self.conn.execute("PRAGMA locking_mode = EXCLUSIVE;")

    def write_batch(self, records: Sequence[Tuple[Any, ...]]) -> None:
        """批次寫入 11 欄晶片明細資料（建議每批 50,000 列）"""
        if not records:
            return
        if self._closed or self.conn is None:
            raise RuntimeError("SidecarCacheWriter 連線已關閉，無法繼續寫入。")

        try:
            self.conn.execute("BEGIN TRANSACTION;")
            self.conn.executemany(self.INSERT_SQL, records)
            self.conn.execute("COMMIT;")
            self.total_rows += len(records)
        except Exception:
            self.abort()
            raise

    def finalize(self, final_xlsx_path: Union[str, Path]) -> Path:
        """
        寫入最終 .xlsx 的 file_size 與 st_mtime_ns 指紋，並將 WAL 合併封口後關閉連線。
        注意：必須在 final_xlsx_path 已完成 wb.close() 並搬移到位後呼叫。
        """
        if self._closed or self.conn is None:
            raise RuntimeError("SidecarCacheWriter 連線已關閉，無法執行 finalize。")

        try:
            stat = os.stat(final_xlsx_path)
            created_at = datetime.now(timezone.utc).isoformat()

            self.conn.execute("BEGIN TRANSACTION;")
            self.conn.execute(
                """
                INSERT OR REPLACE INTO meta (id, file_size, mtime_ns, row_count, created_at)
                VALUES (1, ?, ?, ?, ?)
                """,
                (stat.st_size, stat.st_mtime_ns, self.total_rows, created_at),
            )
            self.conn.execute("COMMIT;")

            # 將 WAL 日誌合併回主資料庫並切回 DELETE 模式，避免 Windows 目錄殘留 -wal/-shm 檔
            self.conn.execute("PRAGMA wal_checkpoint(TRUNCATE);")
            self.conn.execute("PRAGMA journal_mode = DELETE;")
            self.conn.close()
            self._closed = True
            return self.cache_path
        except Exception:
            self.abort()
            raise

    def abort(self) -> None:
        """異常發生時安全回滾、釋放控制代碼並刪除殘留的髒快取檔案"""
        if self.conn is not None:
            try:
                if self.conn.in_transaction:
                    self.conn.rollback()
            except sqlite3.Error:
                pass
            try:
                self.conn.close()
            except sqlite3.Error:
                pass
            self.conn = None
        self._closed = True
        self._cleanup_files()

    def _cleanup_files(self) -> None:
        for suffix in ("", "-wal", "-shm", "-journal"):
            p = Path(f"{self.cache_path}{suffix}")
            try:
                if p.exists():
                    p.unlink()
            except OSError:
                pass

    def __enter__(self) -> "SidecarCacheWriter":
        return self

    def __exit__(self, exc_type: Any, exc_val: Any, exc_tb: Any) -> None:
        if exc_type is not None:
            self.abort()
        elif not self._closed:
            self.abort()


class SidecarCacheReader:
    """彙整 2 專用：校驗旁路快取有效性、秒撈不重複 RECIPE 與串流讀取晶片資料"""

    @staticmethod
    def is_valid(xlsx_path: Union[str, Path], cache_path: Optional[Union[str, Path]] = None) -> bool:
        xlsx_p = Path(xlsx_path)
        cache_p = Path(cache_path) if cache_path else get_cache_path(xlsx_p)

        if not xlsx_p.exists() or not cache_p.exists():
            return False

        try:
            stat = os.stat(xlsx_p)
            conn = sqlite3.connect(f"file:{cache_p}?mode=ro", uri=True)
            try:
                row = conn.execute(
                    "SELECT file_size, mtime_ns FROM meta WHERE id = 1"
                ).fetchone()
                return bool(
                    row is not None
                    and int(row[0]) == stat.st_size
                    and int(row[1]) == stat.st_mtime_ns
                )
            finally:
                conn.close()
        except (OSError, sqlite3.Error):
            return False

    @staticmethod
    def get_distinct_recipes(cache_path: Union[str, Path]) -> List[str]:
        """在 0.05 秒內透過 SQL 撈出所有唯一的非空 RECIPE 清單（排序確保決定性）"""
        conn = sqlite3.connect(f"file:{cache_path}?mode=ro", uri=True)
        try:
            rows = conn.execute("SELECT DISTINCT recipe FROM wafer_details").fetchall()
            recipes = sorted({str(r[0]).strip() for r in rows if r and r[0] is not None and str(r[0]).strip()})
            return recipes
        finally:
            conn.close()

    @staticmethod
    def has_blank_recipe(cache_path: Union[str, Path]) -> bool:
        """快速檢查是否存在空白 RECIPE（決定是否需要建立 ERROR_blank 欄位）"""
        conn = sqlite3.connect(f"file:{cache_path}?mode=ro", uri=True)
        try:
            row = conn.execute(
                "SELECT 1 FROM wafer_details WHERE recipe IS NULL OR TRIM(recipe) = '' LIMIT 1"
            ).fetchone()
            return row is not None
        finally:
            conn.close()

    @staticmethod
    def stream_records(
        cache_path: Union[str, Path], batch_size: int = 50_000
    ) -> Iterator[Tuple[Any, ...]]:
        """以固定批次大小串流讀取 11 欄明細，記憶體維持恆定"""
        conn = sqlite3.connect(f"file:{cache_path}?mode=ro", uri=True)
        try:
            conn.execute("PRAGMA cache_size = 50000;")
            cursor = conn.execute(
                "SELECT layer1, pn, layer2, lot, recipe, file_name, "
                "mpid, mp_x, mp_y, mpid_x_y, unit_bin FROM wafer_details"
            )
            while True:
                batch = cursor.fetchmany(batch_size)
                if not batch:
                    break
                yield from batch
        finally:
            conn.close()
```

## 模組二：彙整 1 引擎雙軌極速串流輸出 (`core/detail_aggregator.py`)

> **正統業務校正重點**：

> 2. 保留第一欄為 `Hard Limit` 之非晶片行排除機制。
> 3. 保留 `aggregate_to_excel(manifest_csv_path, output_xlsx_path, log_cb, progress_cb)` 之既有公開介面簽名與回傳元組 `(succeeded, failed, total_written)`。

```python
# core/detail_aggregator.py
import csv
import os
import tempfile
from pathlib import Path
from typing import Any, Callable, Dict, List, Optional, Sequence, Tuple
import xlsxwriter

from core.excel_utils import safe_atomic_move, sanitize_xml_text
from core.sidecar_cache import SidecarCacheWriter


class DetailAggregator:
    """
    彙整資料 1（極速雙軌串流版）：
    - 管道 A：xlsxwriter(constant_memory=True) 寫入 %TEMP% 暫存檔，結束後安全原子搬移
    - 管道 B：同步每 50,000 列寫入 SQLite 旁路快取 (.cache.db)，供彙整 2 秒級讀取
    - 完美相容既有介面簽名：aggregate_to_excel(...) -> (succeeded, failed, total_written)
    """

    OUT_SHEET_NAME = "工作表2"
    MAX_ROWS_PER_SHEET = 1_000_000  # 含表頭列
    CACHE_BATCH_SIZE = 50_000

    DETAIL_HEADERS: List[str] = [
        "第 1 層目錄",
        "PN",
        "第 2 層目錄",
        "LOT",
        "RECIPE",
        "檔案名稱",
        "2D ID (MPID)",
        "X (MP_X)",
        "Y (MP_Y)",
        "MPID_X_Y",
        "Unit BIN#",
    ]

    # A~J 欄寬度 (索引 0~9 鎖定純文字 '@')，K 欄 (索引 10) 寬度 13 保留 General/數值格式
    COLUMN_WIDTHS: Dict[str, int] = {
        "A": 24, "B": 13, "C": 22, "D": 22, "E": 12,
        "F": 62, "G": 21, "H": 13, "I": 13, "J": 28, "K": 13
    }
    DETAIL_COL_WIDTHS_LIST: Tuple[int, ...] = (24, 13, 22, 22, 12, 62, 21, 13, 13, 28, 13)

    @staticmethod
    def parse_recipe(level1_dir: str, pn: str = "") -> str:
        """
        RECIPE 解析演算法 (方案 2: 以 PN 為錨點自動定位 + 連字號自適應退避):
        1. 優先方案: 若有提供 pn 且存在於 level1_dir:
           - 找到 pn 在 level1_dir 中的位置
           - 從 pn 之後尋找第一個連字號 '-'
           - 取該連字號之後的全部文字作為 RECIPE (例如 '214-3273242A-SUB-C' 配 '3273242' -> 'SUB-C')
           - 亦涵蓋無廠區前綴結構 (例如 '3273173A-DUMMY' 配 '3273173' -> 'DUMMY')
        2. 退避方案: 若無 pn 或 pn 未在目錄名中命中:
           - 若有 >= 2 個連字號: 取第 2 個連字號後的全部文字
           - 若只有 1 個連字號: 取第 1 個連字號後的全部文字
           - 若無連字號: 回傳空字串
        """
        s = level1_dir.strip()
        p = pn.strip()

        # 優先方案: 以 PN 為錨點
        if p and p.lower() in s.lower():
            pn_pos = s.lower().find(p.lower())
            dash_pos = s.find("-", pn_pos + len(p))
            if dash_pos != -1:
                recipe = s[dash_pos + 1:].strip()
                if recipe:
                    return recipe

        # 退避方案: 連字號數量自適應
        dashes = [i for i, ch in enumerate(s) if ch == "-"]
        if len(dashes) >= 2:
            return s[dashes[1] + 1:].strip()
        elif len(dashes) == 1:
            return s[dashes[0] + 1:].strip()
        return ""

    @staticmethod
    def read_local_text(file_path: str) -> str:
        """支援 UTF-8 BOM、標準 UTF-8 及 Windows CP950/ANSI 編碼自動辨識讀取"""
        with open(file_path, "rb") as f:
            raw_bytes = f.read()

        if not raw_bytes:
            return ""

        # 1. 檢查 UTF-8 BOM
        if raw_bytes.startswith(b"\xef\xbb\xbf"):
            return raw_bytes[3:].decode("utf-8", errors="replace")

        # 2. 嘗試標準 UTF-8
        try:
            return raw_bytes.decode("utf-8")
        except UnicodeDecodeError:
            pass

        # 3. 嘗試 Windows 繁體中文 CP950 (ANSI)
        try:
            return raw_bytes.decode("cp950")
        except UnicodeDecodeError:
            pass

        # 4. 保底 Latin-1
        return raw_bytes.decode("latin-1", errors="replace")

    @classmethod
    def parse_single_file(
        cls,
        file_path: str,
        level1: str,
        pn: str,
        level2: str,
        lot: str
    ) -> Tuple[List[List[Any]], Optional[str]]:
        """
        解析單一晶片測試檔案：
        - 搜尋前 10 列中包含 2D ID, X, Y, UNIT BIN# 之標題列（大小寫不敏感，相容逗號與 Tab）
        - 第一欄為 Hard Limit 之列略過不計
        - RECIPE 依 PN 錨點解析
        """
        if not os.path.exists(file_path):
            return [], f"實體檔案不存在: {file_path}"

        raw_text = cls.read_local_text(file_path)
        if not raw_text.strip():
            return [], "空檔案"

        lines = [line.strip("\r\n") for line in raw_text.splitlines() if line.strip("\r\n")]
        if not lines:
            return [], "無有效內容"

        # 尋找前 10 列中的標題列
        header_idx = -1
        sep = ","
        id_col = -1
        x_col = -1
        y_col = -1
        bin_col = -1

        for i, line in enumerate(lines[:10]):
            line_upper = line.upper()
            if "2D ID" in line_upper:
                sep = "\t" if "\t" in line else ","
                reader = csv.reader([line], delimiter=sep)
                fields = next(reader, [])
                t_id, t_x, t_y, t_bin = -1, -1, -1, -1
                for k, col in enumerate(fields):
                    c_clean = col.strip().upper()
                    if c_clean == "2D ID":
                        t_id = k
                    elif c_clean == "X":
                        t_x = k
                    elif c_clean == "Y":
                        t_y = k
                    elif c_clean in ("UNIT BIN#", "UNIT BIN #", "BIN#"):
                        t_bin = k
                if t_id >= 0 and t_x >= 0 and t_y >= 0 and t_bin >= 0:
                    header_idx = i
                    id_col, x_col, y_col, bin_col = t_id, t_x, t_y, t_bin
                    break

        if header_idx < 0:
            return [], "前 10 列找不到 2D ID、X、Y、Unit BIN# 欄位"

        file_name = os.path.basename(file_path)
        sub_code = cls.parse_recipe(level1, pn)
        max_col_needed = max(id_col, x_col, y_col, bin_col)

        rows: List[List[Any]] = []
        for line in lines[header_idx + 1:]:
            if not line.strip():
                continue
            reader = csv.reader([line], delimiter=sep)
            cols = next(reader, [])
            if not cols:
                continue

            # 略過 Hard Limit 列
            first_field = cols[0].strip()
            if first_field.lower() == "hard limit":
                continue

            if len(cols) > max_col_needed:
                unit_id = cols[id_col].strip()
                x_val = cols[x_col].strip()
                y_val = cols[y_col].strip()
                bin_val = cols[bin_col].strip()

                if unit_id or bin_val:
                    mpid_x_y = f"{unit_id}_{x_val}_{y_val}" if (unit_id or x_val or y_val) else ""
                    final_bin: Any = bin_val
                    if bin_val.isdigit():
                        try:
                            final_bin = int(bin_val)
                        except ValueError:
                            final_bin = bin_val

                    row = [
                        level1,
                        pn,
                        level2,
                        lot,
                        sub_code,
                        file_name,
                        unit_id,
                        x_val,
                        y_val,
                        mpid_x_y,
                        final_bin
                    ]
                    rows.append(row)

        return rows, None

    @classmethod
    def _setup_detail_sheet(
        cls,
        ws: xlsxwriter.worksheet.Worksheet,
        fmt_text: xlsxwriter.format.Format,
        fmt_header_text: xlsxwriter.format.Format,
        fmt_header_general: xlsxwriter.format.Format,
    ) -> None:
        """在寫入資料列前鎖定欄寬、純文字格式 (@) 與表頭列（Row 0）"""
        for col_idx, width in enumerate(cls.DETAIL_COL_WIDTHS_LIST):
            if col_idx <= 9:  # A ~ J 欄：純文字格式 '@'
                ws.set_column(col_idx, col_idx, width, fmt_text)
            else:  # K 欄 (Unit BIN#)：保留 General / 數值格式
                ws.set_column(col_idx, col_idx, width)

        for col_idx, header_text in enumerate(cls.DETAIL_HEADERS):
            hdr_fmt = fmt_header_text if col_idx <= 9 else fmt_header_general
            ws.write_string(0, col_idx, header_text, hdr_fmt)

    @staticmethod
    def _write_bin_cell(
        ws: xlsxwriter.worksheet.Worksheet, row_idx: int, col_idx: int, bin_val: Any
    ) -> None:
        """K 欄 (Unit BIN#) 寫入：數值以 write_number 輸出，字串以 write_string 輸出"""
        if bin_val is None or bin_val == "":
            return
        if isinstance(bin_val, int):
            ws.write_number(row_idx, col_idx, bin_val)
        elif isinstance(bin_val, str) and bin_val.isdigit() and (len(bin_val) == 1 or not bin_val.startswith("0")):
            ws.write_number(row_idx, col_idx, int(bin_val))
        else:
            ws.write_string(row_idx, col_idx, str(bin_val))

    @classmethod
    def aggregate_to_excel(
        cls,
        manifest_csv_path: str,
        output_xlsx_path: str,
        log_cb: Optional[Callable[[str], None]] = None,
        progress_cb: Optional[Callable[[int, int, int, str], None]] = None
    ) -> Tuple[int, int, int]:
        """
        讀取下載清單 CSV，解析其所有「下載成功」的檔案，透過雙軌串流產出：
        - 管道 A：xlsxwriter(constant_memory=True) 輸出雙工作表 Excel（Sheet 1 下載清單、Sheet 2 工作表2...）
        - 管道 B：同步產生 SQLite 旁路快取 (.cache.db)，供彙整 2 秒級載入
        回傳: (成功解析檔案數, 略過/失敗檔案數, 總寫入明細筆數)
        """
        if not os.path.exists(manifest_csv_path):
            raise FileNotFoundError(f"找不到下載清單: {manifest_csv_path}")

        # 1. 讀取下載清單 CSV
        with open(manifest_csv_path, "r", encoding="utf-8-sig") as f:
            reader = list(csv.reader(f))

        if not reader:
            raise ValueError("下載清單為空檔案")

        manifest_headers = reader[0]
        manifest_rows = reader[1:]

        col_map = {name.strip(): idx for idx, name in enumerate(manifest_headers)}
        path_col = col_map.get("本地完整路徑")
        status_col = col_map.get("處理狀態")
        l1_col = col_map.get("第 1 層目錄")
        pn_col = col_map.get("第 1 層搜尋字")
        l2_col = col_map.get("第 2 層目錄")
        lot_col = col_map.get("第 2 層搜尋字")

        if path_col is None or status_col is None:
            raise ValueError("下載清單缺少必要欄位：'本地完整路徑' 或 '處理狀態'")

        valid_items = []
        for r_idx, r in enumerate(manifest_rows, start=2):
            local_path = r[path_col].strip() if path_col < len(r) else ""
            status = r[status_col].strip() if status_col < len(r) else ""
            if status == "下載成功" and local_path:
                valid_items.append((r_idx, r, local_path))

        total_files = len(valid_items)
        tmp_dir = tempfile.gettempdir()
        fd, temp_xlsx_path = tempfile.mkstemp(prefix="wafer_detail_", suffix=".xlsx", dir=tmp_dir)
        os.close(fd)

        cache_writer = SidecarCacheWriter(output_xlsx_path)
        cache_buffer: List[Tuple[Any, ...]] = []

        wb = xlsxwriter.Workbook(
            temp_xlsx_path,
            {"constant_memory": True, "tmpdir": tmp_dir},
        )
        wb_closed = False

        try:
            fmt_list_header = wb.add_format({"bold": True})
            fmt_text = wb.add_format({"num_format": "@"})
            fmt_header_text = wb.add_format({"bold": True, "num_format": "@"})
            fmt_header_general = wb.add_format({"bold": True})

            # Sheet 1: 下載清單（表頭加粗、欄寬 18）
            ws_list = wb.add_worksheet("下載清單")
            num_list_cols = max(len(manifest_headers), 1)
            ws_list.set_column(0, num_list_cols - 1, 18)

            for c_idx, h_val in enumerate(manifest_headers):
                ws_list.write_string(0, c_idx, sanitize_xml_text(h_val), fmt_list_header)

            for r_offset, row_vals in enumerate(manifest_rows, start=1):
                for c_idx, val in enumerate(row_vals):
                    if val is not None and val != "":
                        ws_list.write_string(r_offset, c_idx, sanitize_xml_text(val))

            # Sheet 2+: 工作表2 (首頁無後綴，超過 100 萬列切 工作表2_2, 工作表2_3...)
            page_no = 1
            current_ws = wb.add_worksheet(cls.OUT_SHEET_NAME)
            cls._setup_detail_sheet(current_ws, fmt_text, fmt_header_text, fmt_header_general)

            row_in_sheet = 1
            total_written = 0
            succeeded_files = 0
            failed_files = 0

            for file_idx, (r_idx, r, local_path) in enumerate(valid_items, start=1):
                l1 = r[l1_col].strip() if l1_col is not None and l1_col < len(r) else ""
                pn = r[pn_col].strip() if pn_col is not None and pn_col < len(r) else ""
                l2 = r[l2_col].strip() if l2_col is not None and l2_col < len(r) else ""
                lot = r[lot_col].strip() if lot_col is not None and lot_col < len(r) else ""

                fname = os.path.basename(local_path)
                parsed_rows, err = cls.parse_single_file(local_path, l1, pn, l2, lot)
                if err:
                    failed_files += 1
                    if log_cb:
                        log_cb(f"[略過] [{file_idx}/{total_files}] 第 {r_idx} 列 {fname}: {err}")
                    if progress_cb:
                        progress_cb(file_idx, total_files, total_written, fname)
                    continue

                succeeded_files += 1

                for row_data in parsed_rows:
                    if row_in_sheet >= cls.MAX_ROWS_PER_SHEET:
                        page_no += 1
                        sheet_name = f"{cls.OUT_SHEET_NAME}_{page_no}"
                        current_ws = wb.add_worksheet(sheet_name)
                        cls._setup_detail_sheet(
                            current_ws, fmt_text, fmt_header_text, fmt_header_general
                        )
                        row_in_sheet = 1

                    # A ~ J 欄寫入純文字
                    for c_idx in range(10):
                        v_str = sanitize_xml_text(row_data[c_idx])
                        if v_str:
                            current_ws.write_string(row_in_sheet, c_idx, v_str)

                    # K 欄 (Unit BIN#)
                    cls._write_bin_cell(current_ws, row_in_sheet, 10, row_data[10])

                    row_in_sheet += 1
                    total_written += 1

                    # 旁路快取累積寫入
                    cache_buffer.append(tuple(sanitize_xml_text(col) for col in row_data))
                    if len(cache_buffer) >= cls.CACHE_BATCH_SIZE:
                        cache_writer.write_batch(cache_buffer)
                        cache_buffer.clear()

                if progress_cb:
                    progress_cb(file_idx, total_files, total_written, fname)

                if log_cb and (file_idx == 1 or file_idx % 5 == 0 or file_idx == total_files):
                    pct = (file_idx / total_files) * 100 if total_files > 0 else 100.0
                    log_cb(f"[{file_idx}/{total_files}] ({pct:.1f}%) 正在解析: {fname} (累計寫入 {total_written:,} 筆晶片)...")

            # 沖刷剩餘快取
            if cache_buffer:
                cache_writer.write_batch(cache_buffer)
                cache_buffer.clear()

            if log_cb:
                log_cb(f"所有檔案解析完成 (共 {succeeded_files} 個成功檔，寫入 {total_written:,} 筆晶片)，正在封裝 Excel 活頁簿至硬碟 (請稍候)...")

            wb.close()
            wb_closed = True

            safe_atomic_move(temp_xlsx_path, output_xlsx_path)
            cache_writer.finalize(output_xlsx_path)

            return succeeded_files, failed_files, total_written

        except Exception:
            if not wb_closed:
                try:
                    wb.close()
                except Exception:
                    pass
            if os.path.exists(temp_xlsx_path):
                try:
                    os.remove(temp_xlsx_path)
                except OSError:
                    pass
            cache_writer.abort()
            raise
```

## 模組三：彙整 2 引擎升級與兩階段極速樞紐轉置 (`core/pivot_aggregator.py`)

> **正統業務校正重點**：
> 1. **維持正統雙層分群**：
>    - `details` 依 `(pn, full_lot)` 分群 $\rightarrow$ 輸出 `中繼明細` 系列分頁。
>    - `summaries` 依 `(pn, clean_lot)` 母批號分群 $\rightarrow$ **同一母批號下的所有晶圓合併在同一列**，B 欄 `第 2 層目錄(LOT)` 逗號串接所有 `full_lot`，跨晶圓取各 RECIPE 最高權重，橫向比較產生 `BinCode_sumary`！
> 2. **智慧路由載入**：優先命中 SQLite 旁路快取（1.5 秒），退避路徑使用 `python-calamine`（15 秒）。
> 3. **保留原公開簽名**：`aggregate_pivot(...) -> (total_source_rows, detail_written, summary_written, output_xlsx_path)`。

```python
# core/pivot_aggregator.py
import gc
import math
import os
import sys
import tempfile
from typing import Any, Callable, Dict, Iterator, List, Optional, Sequence, Tuple
import xlsxwriter
from python_calamine import CalamineWorkbook

from core.excel_utils import safe_atomic_move, sanitize_xml_text
from core.sidecar_cache import SidecarCacheReader, get_cache_path


def sanitize_bin_value(val: Any) -> str:
    """清洗 Unit BIN# 值，確保輸出為乾淨純字串"""
    if val is None or isinstance(val, bool):
        return ""
    if isinstance(val, int):
        return str(val)
    if isinstance(val, float):
        if math.isnan(val) or math.isinf(val):
            return ""
        return str(int(val)) if val.is_integer() else str(val).strip()
    return sanitize_xml_text(val).strip()


def sanitize_text_cell(val: Any) -> str:
    """一般欄位安全轉為乾淨字串"""
    if val is None:
        return ""
    if isinstance(val, float):
        if math.isnan(val) or math.isinf(val):
            return ""
        return str(int(val)) if val.is_integer() else str(val).strip()
    return sanitize_xml_text(val).strip()


def pad_coordinates(mpid_xy: Any) -> str:
    """
    晶圓座標補零正規化：
    將 MPID_X_Y 尾端兩碼純數字自動補零為至少兩位數（例如 'ID_1_2' -> 'ID_01_02'）。
    """
    s = sanitize_text_cell(mpid_xy)
    if not s:
        return ""
    parts = s.split("_")
    if len(parts) >= 3 and parts[-1].isdigit() and parts[-2].isdigit():
        parts[-2] = parts[-2].zfill(2)
        parts[-1] = parts[-1].zfill(2)
        return "_".join(parts)
    if len(parts) == 2 and parts[0].isdigit() and parts[1].isdigit():
        return f"{parts[0].zfill(2)}_{parts[1].zfill(2)}"
    return s


class PaginatedStreamSheetWriter:
    """
    基於 xlsxwriter(constant_memory=True) 的百萬列自動切頁串流寫入器：
    - 支援群組完整性保護（整包 LOT 若可放入新頁則不在舊頁尾端拆碎）
    - 切頁時自動建立新工作表 (_01, _02...)、設定欄寬與純文字格式 (@)、重繪深藍/深橘表頭
    """

    def __init__(
        self,
        workbook: xlsxwriter.Workbook,
        base_sheet_name: str,
        fixed_headers: Sequence[str],
        dynamic_recipes: Sequence[str],
        has_error_blank: bool,
        col_widths: Sequence[float],
        max_rows_per_sheet: int = 1_000_000,
    ):
        self.wb = workbook
        self.base_name = base_sheet_name
        self.fixed_headers = list(fixed_headers)
        self.dynamic_recipes = list(dynamic_recipes)
        self.has_error_blank = has_error_blank
        self.col_widths = list(col_widths)
        self.max_rows_per_sheet = max_rows_per_sheet

        self.fmt_text = self.wb.add_format({"num_format": "@"})
        self.fmt_header_blue = self.wb.add_format({
            "bold": True,
            "font_color": "#FFFFFF",
            "bg_color": "#1F4E78",  # RGB(31, 78, 120) 深藍底白字
            "num_format": "@",
            "align": "center",
            "valign": "vcenter",
        })
        self.fmt_header_orange = self.wb.add_format({
            "bold": True,
            "font_color": "#FFFFFF",
            "bg_color": "#C65911",  # RGB(198, 89, 17) 深橘底白字 (ERROR_blank)
            "num_format": "@",
            "align": "center",
            "valign": "vcenter",
        })

        self.page_index = 0
        self.current_row = 0
        self.ws: Optional[xlsxwriter.worksheet.Worksheet] = None
        self.first_worksheet: Optional[xlsxwriter.worksheet.Worksheet] = None
        self.total_written_rows = 0
        self._open_new_sheet()

    def _open_new_sheet(self) -> None:
        self.page_index += 1
        sheet_title = f"{self.base_name}_{self.page_index:02d}"
        self.ws = self.wb.add_worksheet(sheet_title)
        if self.first_worksheet is None:
            self.first_worksheet = self.ws
        self.current_row = 0

        total_cols = (
            len(self.fixed_headers)
            + len(self.dynamic_recipes)
            + (1 if self.has_error_blank else 0)
        )
        for col_idx in range(total_cols):
            width = self.col_widths[col_idx] if col_idx < len(self.col_widths) else 14
            self.ws.set_column(col_idx, col_idx, width, self.fmt_text)

        col_cursor = 0
        for h in self.fixed_headers:
            self.ws.write_string(0, col_cursor, h, self.fmt_header_blue)
            col_cursor += 1

        for recipe in self.dynamic_recipes:
            self.ws.write_string(0, col_cursor, recipe, self.fmt_header_blue)
            col_cursor += 1

        if self.has_error_blank:
            self.ws.write_string(0, col_cursor, "ERROR_blank", self.fmt_header_orange)

        self.current_row = 1

    def write_lot_group(self, group_rows: Sequence[Sequence[str]]) -> None:
        """寫入單一 LOT 晶片群組，落實群組完整性保護與自動切頁"""
        group_len = len(group_rows)
        if group_len == 0:
            return

        fits_in_fresh_sheet = (1 + group_len) <= self.max_rows_per_sheet
        if (
            (self.current_row + group_len > self.max_rows_per_sheet)
            and self.current_row > 1
            and fits_in_fresh_sheet
        ):
            self._open_new_sheet()

        for row_vals in group_rows:
            if self.current_row >= self.max_rows_per_sheet:
                self._open_new_sheet()

            assert self.ws is not None
            for col_idx, val in enumerate(row_vals):
                if val:
                    self.ws.write_string(self.current_row, col_idx, val)
            self.current_row += 1
            self.total_written_rows += 1


class PivotAggregator:
    """
    彙整資料 2（PN/LOT/RECIPE 交叉樞紐分析與權重評級）：
    - 優先命中 SQLite 旁路快取 (~1.8 秒)；跨機僅有單一 .xlsx 時自動退避至 Rust calamine (~6 秒)
    - 嚴格落實正統雙層分群：
      1. details: (pn, full_lot) 群組 -> 中繼明細
      2. summaries: (pn, clean_lot) 母批號群組 -> RECIPE彙整 (多晶圓合併同一列、B欄逗號串接、跨晶圓最高權重比較)
    - 完美相容既有介面簽名：aggregate_pivot(...) -> (total_source_rows, detail_written, summary_written, output_xlsx_path)
    """

    MAX_ROWS_PER_SHEET = 1_000_000
    BLANK_HEADER = "ERROR_blank"

    REQUIRED_COLS: Tuple[str, ...] = (
        "第 1 層目錄",
        "PN",
        "第 2 層目錄",
        "LOT",
        "RECIPE",
        "檔案名稱",
        "2D ID (MPID)",
        "X (MP_X)",
        "Y (MP_Y)",
        "MPID_X_Y",
        "Unit BIN#",
    )

    @staticmethod
    def calc_bin_rank(codes: str) -> int:
        """BIN 優先權判定演算法: 包含 32 -> 3; 包含 1 -> 2; 其他數值 -> 1; 空 -> 0"""
        if not codes:
            return 0
        tokens = [c.strip() for c in codes.split(",") if c.strip()]
        if not tokens:
            return 0
        token_set = set(tokens)
        if "32" in token_set:
            return 3
        if "1" in token_set:
            return 2
        return 1

    @staticmethod
    def display_rank(priority: int, whole_row: bool = False) -> Any:
        if priority == 3:
            return 32
        elif priority == 2:
            return 1
        elif priority == 1:
            return "Others" if whole_row else "others"
        return ""

    @staticmethod
    def extract_lot(clean_lot: str, full_lot: str) -> str:
        c = clean_lot.strip()
        if c:
            return c
        f = full_lot.strip()
        if len(f) > 1:
            return f[:-1]
        return f

    @staticmethod
    def extract_pn(clean_pn: str, level1_dir: str) -> str:
        p = clean_pn.strip()
        if p:
            return p
        return level1_dir.strip()

    @classmethod
    def stream_source_records(
        cls, source_xlsx_path: str, log_cb: Optional[Callable[[str], None]] = None
    ) -> Tuple[str, Iterator[Tuple[Any, ...]]]:
        """智慧路由資料載入器：回傳 ('sqlite_cache' | 'calamine_fallback', iterator)"""
        cache_path = get_cache_path(source_xlsx_path)
        if SidecarCacheReader.is_valid(source_xlsx_path, cache_path):
            if log_cb:
                log_cb("偵測到有效 SQLite 旁路快取，啟動極速二進位串流模式...")
            return "sqlite_cache", SidecarCacheReader.stream_records(cache_path)

        if log_cb:
            log_cb("未偵測到旁路快取（或主檔已變更），自動啟動 Rust calamine 退避解析引擎...")

        def _calamine_generator() -> Iterator[Tuple[Any, ...]]:
            wb = CalamineWorkbook.from_path(source_xlsx_path)
            for sheet_name in wb.sheet_names:
                rows_iter = iter(
                    wb.get_sheet_by_name(sheet_name).to_python(skip_empty_area=True)
                )
                header = next(rows_iter, None)
                if not header or sanitize_text_cell(header[0]) != "第 1 層目錄":
                    continue

                col_map = {
                    sanitize_text_cell(h): idx
                    for idx, h in enumerate(header)
                    if h is not None and sanitize_text_cell(h)
                }
                indices = [col_map.get(col, -1) for col in cls.REQUIRED_COLS]

                for row in rows_iter:
                    row_len = len(row)
                    yield tuple(
                        row[idx] if (0 <= idx < row_len) else None
                        for idx in indices
                    )

        return "calamine_fallback", _calamine_generator()

    @classmethod
    def aggregate_pivot(
        cls,
        source_xlsx_path: str,
        output_xlsx_path: Optional[str] = None,
        log_cb: Optional[Callable[[str], None]] = None,
        progress_cb: Optional[Callable[[float, str], None]] = None,
        max_rows_per_sheet: int = MAX_ROWS_PER_SHEET,
    ) -> Tuple[int, int, int, str]:
        """
        核心執行進入點 (完美相容既有回傳規格):
        回傳: (來源有效列數, 中繼明細寫入列數, RECIPE彙整寫入列數, 產出檔案路徑)
        """
        if not os.path.exists(source_xlsx_path):
            raise FileNotFoundError(f"找不到來源活頁簿: {source_xlsx_path}")

        if not output_xlsx_path:
            dir_name = os.path.dirname(os.path.abspath(source_xlsx_path))
            import datetime
            ts = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
            output_xlsx_path = os.path.join(dir_name, f"PN_LOT_RECIPE_明細與彙整_{ts}.xlsx")

        source_mode, records_iter = cls.stream_source_records(source_xlsx_path, log_cb=log_cb)
        cache_path = get_cache_path(source_xlsx_path)

        # -------------------------------------------------------------
        # 階段一：RECIPE 欄位預探索
        # -------------------------------------------------------------
        recipe_to_id: Dict[str, int] = {}
        id_to_recipe: List[str] = []
        has_error_blank = False

        if source_mode == "sqlite_cache":
            for rec in SidecarCacheReader.get_distinct_recipes(cache_path):
                interned_rec = sys.intern(rec)
                recipe_to_id[interned_rec] = len(id_to_recipe)
                id_to_recipe.append(interned_rec)
            has_error_blank = SidecarCacheReader.has_blank_recipe(cache_path)

        ERROR_BLANK_ID = -1

        # 正統雙層分群資料結構：
        # details[pn][full_lot] = {"lot": lot, "ids": [mpid], "id_set": {mpid: idx}, "bins": {rec_id: {idx: list[bin]}}}
        # summaries[pn][lot] = {"ids": [mpid], "id_set": {mpid: idx}, "full_lists": {idx: [full_lot...]}, "bins": {rec_id: {idx: max_rank}}, "row_ranks": {idx: max_row_rank}}
        details: Dict[str, Dict[str, Dict[str, Any]]] = {}
        summaries: Dict[str, Dict[str, Dict[str, Any]]] = {}

        total_source_rows = 0
        was_gc_enabled = gc.isenabled()
        if was_gc_enabled:
            gc.disable()

        try:
            for row in records_iter:
                total_source_rows += 1
                if progress_cb and total_source_rows % 25_000 == 0:
                    progress_cb(0.1, f"讀取與分群聚合: {total_source_rows:,} 列")

                l1 = sanitize_text_cell(row[0])
                pn_raw = sanitize_text_cell(row[1])
                l2 = sanitize_text_cell(row[2])
                lot_raw = sanitize_text_cell(row[3])
                recipe_raw = sanitize_text_cell(row[4])
                mpid = sanitize_text_cell(row[6])
                mp_x = sanitize_text_cell(row[7])
                mp_y = sanitize_text_cell(row[8])
                mpid_xy_raw = sanitize_text_cell(row[9])
                unit_bin = sanitize_bin_value(row[10])

                pn = sys.intern(cls.extract_pn(pn_raw, l1))
                full_lot = sys.intern(l2)
                lot = sys.intern(cls.extract_lot(lot_raw, full_lot))

                if not mpid_xy_raw and (mpid or mp_x or mp_y):
                    mpid_xy_raw = f"{mpid}_{mp_x}_{mp_y}"
                mpid_xy = pad_coordinates(mpid_xy_raw)
                if not mpid_xy:
                    continue

                if not recipe_raw:
                    has_error_blank = True
                    rec_id = ERROR_BLANK_ID
                else:
                    rec_intern = sys.intern(recipe_raw)
                    rec_id = recipe_to_id.get(rec_intern)  # type: ignore[assignment]
                    if rec_id is None:
                        rec_id = len(id_to_recipe)
                        recipe_to_id[rec_intern] = rec_id
                        id_to_recipe.append(rec_intern)

                # 1. 填入 details 群組（以 full_lot 為鍵）
                if pn not in details:
                    details[pn] = {}
                lots_map = details[pn]
                if full_lot not in lots_map:
                    lots_map[full_lot] = {"lot": lot, "ids": [], "id_set": {}, "bins": {}}
                grp = lots_map[full_lot]

                if mpid_xy not in grp["id_set"]:
                    item_idx = len(grp["ids"])
                    grp["ids"].append(mpid_xy)
                    grp["id_set"][mpid_xy] = item_idx
                else:
                    item_idx = grp["id_set"][mpid_xy]

                if rec_id not in grp["bins"]:
                    grp["bins"][rec_id] = {}
                recipe_bins = grp["bins"][rec_id]

                if item_idx not in recipe_bins:
                    recipe_bins[item_idx] = [unit_bin] if unit_bin else []
                else:
                    if unit_bin and unit_bin not in recipe_bins[item_idx]:
                        recipe_bins[item_idx].append(unit_bin)

                # 2. 填入 summaries 群組（以母批號 clean_lot 為鍵，跨晶圓合併！）
                if pn not in summaries:
                    summaries[pn] = {}
                sum_lots = summaries[pn]
                if lot not in sum_lots:
                    sum_lots[lot] = {
                        "ids": [],
                        "id_set": {},
                        "full_lists": {},
                        "bins": {},
                        "row_ranks": {}
                    }
                s_grp = sum_lots[lot]

                if mpid_xy not in s_grp["id_set"]:
                    s_idx = len(s_grp["ids"])
                    s_grp["ids"].append(mpid_xy)
                    s_grp["id_set"][mpid_xy] = s_idx
                    s_grp["full_lists"][s_idx] = [full_lot]
                else:
                    s_idx = s_grp["id_set"][mpid_xy]
                    if full_lot not in s_grp["full_lists"][s_idx]:
                        s_grp["full_lists"][s_idx].append(full_lot)

                # 權重評級
                if unit_bin:
                    priority = cls.calc_bin_rank(unit_bin)
                    if priority > 0:
                        if rec_id not in s_grp["bins"]:
                            s_grp["bins"][rec_id] = {}
                        dst_hits = s_grp["bins"][rec_id]
                        if s_idx not in dst_hits or priority > dst_hits[s_idx]:
                            dst_hits[s_idx] = priority

                        if rec_id != ERROR_BLANK_ID:
                            if s_idx not in s_grp["row_ranks"] or priority > s_grp["row_ranks"][s_idx]:
                                s_grp["row_ranks"][s_idx] = priority
        finally:
            if was_gc_enabled:
                gc.enable()
            gc.collect()

        if total_source_rows == 0 or not details:
            raise ValueError("來源工作表中無任何有效明細資料可供彙整！")

        # 排序 RECIPE 欄位
        if source_mode == "calamine_fallback":
            recipes_ordered = sorted(id_to_recipe)
            old_to_new_col = {recipe_to_id[rec]: new_idx for new_idx, rec in enumerate(recipes_ordered)}
        else:
            recipes_ordered = id_to_recipe
            old_to_new_col = {idx: idx for idx in range(len(recipes_ordered))}

        num_recipes = len(recipes_ordered)
        if log_cb:
            log_cb(f"來源讀取完成：共 {total_source_rows:,} 列，辨識出 {num_recipes} 個 RECIPE" + ("（含 ERROR_blank）" if has_error_blank else ""))

        # -------------------------------------------------------------
        # 階段二：串流分頁寫出（中繼明細 & RECIPE彙整）
        # -------------------------------------------------------------
        tmp_dir = tempfile.gettempdir()
        fd, temp_xlsx_path = tempfile.mkstemp(prefix="wafer_pivot_", suffix=".xlsx", dir=tmp_dir)
        os.close(fd)

        wb = xlsxwriter.Workbook(
            temp_xlsx_path,
            {"constant_memory": True, "tmpdir": tmp_dir},
        )
        wb_closed = False

        try:
            # 1. 輸出中繼明細系列分頁
            if log_cb:
                log_cb("正在產出 [中繼明細] 工作表...")
            if progress_cb:
                progress_cb(0.4, "正在產出中繼明細工作表...")

            detail_fixed_headers = ["PN", "第 2 層目錄(LOT)", "LOT", "MPID_XY"]
            detail_col_widths = [14, 24, 20, 30] + [18] * (num_recipes + (1 if has_error_blank else 0))
            detail_writer = PaginatedStreamSheetWriter(
                workbook=wb,
                base_sheet_name="中繼明細",
                fixed_headers=detail_fixed_headers,
                dynamic_recipes=recipes_ordered,
                has_error_blank=has_error_blank,
                col_widths=detail_col_widths,
                max_rows_per_sheet=max_rows_per_sheet,
            )

            tail_cols = num_recipes + (1 if has_error_blank else 0)

            for pn, lots_map in details.items():
                for full_lot, grp in lots_map.items():
                    lot = grp["lot"]
                    ids = grp["ids"]
                    if not ids:
                        continue
                    group_rows: List[List[str]] = []
                    for idx, mpid_xy in enumerate(ids):
                        tail = [""] * tail_cols
                        for rec_id, code_dict in grp["bins"].items():
                            c_list = code_dict.get(idx, [])
                            joined = ",".join(c_list) if c_list else ""
                            if rec_id == ERROR_BLANK_ID:
                                if has_error_blank:
                                    tail[-1] = joined
                            else:
                                offset = old_to_new_col[rec_id]
                                tail[offset] = joined
                        group_rows.append([pn, full_lot, lot, mpid_xy] + tail)
                    detail_writer.write_lot_group(group_rows)

            detail_written = detail_writer.total_written_rows

            # 2. 輸出 RECIPE彙整 系列分頁
            if log_cb:
                log_cb("正在產出 [RECIPE彙整] 交叉樞紐表...")
            if progress_cb:
                progress_cb(0.7, "正在產出 RECIPE 彙整交叉表...")

            summary_fixed_headers = ["PN", "第 2 層目錄(LOT)", "LOT", "MPID_XY", "BinCode_sumary"]
            summary_col_widths = [14, 46, 20, 30, 21] + [18] * (num_recipes + (1 if has_error_blank else 0))
            summary_writer = PaginatedStreamSheetWriter(
                workbook=wb,
                base_sheet_name="RECIPE彙整",
                fixed_headers=summary_fixed_headers,
                dynamic_recipes=recipes_ordered,
                has_error_blank=has_error_blank,
                col_widths=summary_col_widths,
                max_rows_per_sheet=max_rows_per_sheet,
            )

            for pn, sum_lots in summaries.items():
                for lot, s_grp in sum_lots.items():
                    ids = s_grp["ids"]
                    if not ids:
                        continue
                    group_rows = []
                    for idx, mpid_xy in enumerate(ids):
                        full_lot_str = ",".join(s_grp["full_lists"].get(idx, []))
                        row_rank = s_grp["row_ranks"].get(idx, 0)
                        summary_bincode = str(cls.display_rank(row_rank, whole_row=True))

                        tail = [""] * tail_cols
                        for rec_id, hit_dict in s_grp["bins"].items():
                            p_val = hit_dict.get(idx, 0)
                            code_str = str(cls.display_rank(p_val, whole_row=False))
                            if rec_id == ERROR_BLANK_ID:
                                if has_error_blank:
                                    tail[-1] = code_str
                            else:
                                offset = old_to_new_col[rec_id]
                                tail[offset] = code_str

                        group_rows.append([pn, full_lot_str, lot, mpid_xy, summary_bincode] + tail)
                    summary_writer.write_lot_group(group_rows)

            summary_written = summary_writer.total_written_rows

            # 預設聚焦於 RECIPE彙整_01
            if summary_writer.first_worksheet is not None:
                summary_writer.first_worksheet.activate()

            if log_cb:
                log_cb(f"所有樞紐運算完成 (中繼明細 {detail_written:,} 列，RECIPE彙整 {summary_written:,} 列)，正在儲存活頁簿至硬碟 (請稍候)...")
            if progress_cb:
                progress_cb(0.9, "正在儲存 Excel 活頁簿至硬碟...")

            wb.close()
            wb_closed = True

            safe_atomic_move(temp_xlsx_path, output_xlsx_path)

            if progress_cb:
                progress_cb(1.0, "樞紐彙整完成！")
            if log_cb:
                log_cb(f"🎉 樞紐彙整完成！中繼明細: {detail_written:,} 列，RECIPE彙整: {summary_written:,} 列")

            return total_source_rows, detail_written, summary_written, output_xlsx_path

        except Exception:
            if not wb_closed:
                try:
                    wb.close()
                except Exception:
                    pass
            if os.path.exists(temp_xlsx_path):
                try:
                    os.remove(temp_xlsx_path)
                except OSError:
                    pass
            raise
```
