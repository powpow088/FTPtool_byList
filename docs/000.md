Option Explicit

Private Const SRC_SHEET As String = "下載清單_20261120_205457"
Private Const OUT_SHEET As String = "工作表2"
Private Const TEST_LIMIT As Long = 0 ' 0 = 全部；3 = 最多嘗試 3 個符合條件的檔案
Private Const BATCH_SIZE As Long = 25000
Private Const OUT_COLS As Long = 11

Private outWs As Worksheet
Private pageNo As Long
Private nextRow As Long
Private buffered As Long
Private written As Long
Private buf() As Variant

' 保留原巨集名稱，供既有按鈕呼叫。
Public Sub 匯入CSV八欄()
    Dim src As Worksheet, ws As Worksheet, clearRange As Range
    Dim pathCol As Long, statusCol As Long
    Dim level1Col As Long, level2Col As Long, pnCol As Long, lotCol As Long
    Dim c As Long, r As Long, lastRow As Long, lastCol As Long
    Dim listData As Variant
    Dim filePath As String, why As String, details As String
    Dim summary As String, prompt As String
    Dim attempted As Long, succeeded As Long, failed As Long, skipped As Long
    Dim oldCalc As XlCalculation, oldEvents As Boolean, oldScreen As Boolean
    Dim oldStatus As Variant, settingsSaved As Boolean, outputStarted As Boolean
    Dim icon As VbMsgBoxStyle

    On Error GoTo RunFailed
    Set src = ThisWorkbook.Worksheets(SRC_SHEET)

    ' 依下載清單的原始標題找欄位，不寫死欄號。
    For c = 1 To src.Cells(1, src.Columns.Count).End(xlToLeft).Column
        Select Case Trim$(CStr(src.Cells(1, c).Value2))
            Case "本地完整路徑": pathCol = c
            Case "處理狀態": statusCol = c
            Case "第 1 層目錄": level1Col = c
            Case "第 1 層搜尋字": pnCol = c
            Case "第 2 層目錄": level2Col = c
            Case "第 2 層搜尋字": lotCol = c
        End Select
    Next c

    If pathCol = 0 Or statusCol = 0 Or level1Col = 0 Or _
       level2Col = 0 Or pnCol = 0 Or lotCol = 0 Then
        MsgBox "下載清單缺少必要欄位：" & vbCrLf & _
               "本地完整路徑、處理狀態、兩層目錄或兩層搜尋字。", vbExclamation
        Exit Sub
    End If

    lastRow = src.Cells(src.Rows.Count, pathCol).End(xlUp).Row
    If lastRow < 2 Then
        MsgBox "下載清單沒有本地檔案路徑。舊輸出未變更。", vbExclamation
        Exit Sub
    End If

    lastCol = Application.Max(pathCol, statusCol, level1Col, level2Col, pnCol, lotCol)
    listData = src.Range(src.Cells(1, 1), src.Cells(lastRow, lastCol)).Value2

    ' 沒有任何待匯入項目時，不清除舊輸出。
    For r = 2 To lastRow
        If Len(Trim$(CStr(listData(r, pathCol)))) > 0 And _
           Trim$(CStr(listData(r, statusCol))) = "下載成功" Then Exit For
    Next r
    If r > lastRow Then
        MsgBox "沒有可匯入的「下載成功」項目。舊輸出未變更。", vbExclamation
        Exit Sub
    End If
    r = 0

    prompt = "將清除「工作表2」及其數字續頁的 A:K，再重新匯入。" & vbCrLf & _
             "下載清單與原始 CSV 不會變更。"
    If TEST_LIMIT > 0 Then
        prompt = prompt & vbCrLf & "注意：測試模式仍會清除上述範圍。"
    End If
    If MsgBox(prompt & vbCrLf & "是否繼續？", _
              vbYesNo Or vbExclamation Or vbDefaultButton2) <> vbYes Then Exit Sub

    oldCalc = Application.Calculation
    oldEvents = Application.EnableEvents
    oldScreen = Application.ScreenUpdating
    oldStatus = Application.StatusBar
    settingsSaved = True

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    outputStarted = True
    For Each ws In ThisWorkbook.Worksheets
        If IsOutputSheet(ws.Name) Then
            Set clearRange = Application.Intersect(ws.UsedRange, ws.Columns("A:K"))
            If Not clearRange Is Nothing Then clearRange.ClearContents
        End If
    Next ws

    pageNo = 1
    buffered = 0
    written = 0
    ReDim buf(1 To BATCH_SIZE, 1 To OUT_COLS)
    PreparePage

    For r = 2 To lastRow
        If TEST_LIMIT > 0 And attempted >= TEST_LIMIT Then Exit For
        filePath = Trim$(CStr(listData(r, pathCol)))

        If Len(filePath) = 0 Or _
           Trim$(CStr(listData(r, statusCol))) <> "下載成功" Then
            skipped = skipped + 1
        Else
            attempted = attempted + 1
            Application.StatusBar = "匯入中：下載清單第 " & r & " / " & lastRow & " 列"

            why = ImportFile(filePath, _
                             CStr(listData(r, level1Col)), CStr(listData(r, pnCol)), _
                             CStr(listData(r, level2Col)), CStr(listData(r, lotCol)))
            If Len(why) = 0 Then
                succeeded = succeeded + 1
            Else
                failed = failed + 1
                Debug.Print "下載清單第 " & r & " 列 | " & filePath & " | " & why
                If failed <= 5 Then
                    details = details & vbCrLf & "第 " & r & " 列：" & why
                End If
            End If
        End If
    Next r

    FlushBuffer
    summary = "匯入結束" & vbCrLf & _
              "成功：" & succeeded & "；失敗：" & failed & "；略過：" & skipped & vbCrLf & _
              "已寫入：" & Format$(written, "#,##0") & " 筆；輸出頁數：" & pageNo & details
    If failed > 5 Then
        summary = summary & vbCrLf & "完整失敗清單：在 VBA 編輯器按 Ctrl+G。"
    End If
    icon = vbInformation
    If failed > 0 Or written = 0 Then icon = vbExclamation
    GoTo Finish

RunFailed:
    summary = "匯入中止：" & Err.Description
    If r >= 2 And r <= lastRow Then
        summary = summary & vbCrLf & "下載清單第 " & r & " 列：" & filePath
    End If
    If outputStarted Then summary = summary & vbCrLf & "目前輸出可能不完整。"
    icon = vbCritical

Finish:
    On Error Resume Next
    If settingsSaved Then
        Application.Calculation = oldCalc
        Application.EnableEvents = oldEvents
        Application.ScreenUpdating = oldScreen
        Application.StatusBar = oldStatus
    End If
    Set outWs = Nothing
    Erase buf
    On Error GoTo 0
    MsgBox summary, icon
End Sub

Private Function IsOutputSheet(ByVal sheetName As String) As Boolean
    Dim suffix As String

    If sheetName = OUT_SHEET Then
        IsOutputSheet = True
    ElseIf Left$(sheetName, Len(OUT_SHEET) + 1) = OUT_SHEET & "_" Then
        suffix = Mid$(sheetName, Len(OUT_SHEET) + 2)
        IsOutputSheet = Len(suffix) > 0 And Not (suffix Like "*[!0-9]*")
    End If
End Function

Private Sub PreparePage()
    Dim sheetName As String

    sheetName = OUT_SHEET
    If pageNo > 1 Then sheetName = sheetName & "_" & pageNo

    Set outWs = Nothing
    On Error Resume Next
    Set outWs = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If outWs Is Nothing Then
        Set outWs = ThisWorkbook.Worksheets.Add( _
            After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        outWs.Name = sheetName
    End If

    With outWs
        .Columns("A:J").NumberFormat = "@"
        .Columns("K").NumberFormat = "General"

        ' 首頁與所有續頁統一使用這組輸出表頭。
        .Range("A1:K1").Value2 = Array( _
            "第 1 層目錄", "PN", "第 2 層目錄", "LOT", "RECIPE", _
            "檔案名稱", "2D ID (MPID)", "X (MP_X)", "Y (MP_Y)", _
            "MPID_X_Y", "Unit BIN#")
        .Range("A1:K1").Font.Bold = True

        .Columns("A").ColumnWidth = 24
        .Columns("B").ColumnWidth = 13
        .Columns("C:D").ColumnWidth = 22
        .Columns("E").ColumnWidth = 12
        .Columns("F").ColumnWidth = 62
        .Columns("G").ColumnWidth = 21
        .Columns("H:I").ColumnWidth = 13
        .Columns("J").ColumnWidth = 28
        .Columns("K").ColumnWidth = 13
    End With
    nextRow = 2
End Sub

Private Function GetSubCode(ByVal level1 As String) As String
    Dim firstDash As Long, secondDash As Long

    level1 = Trim$(level1)
    firstDash = InStr(1, level1, "-", vbBinaryCompare)
    If firstDash = 0 Then Exit Function

    secondDash = InStr(firstDash + 1, level1, "-", vbBinaryCompare)
    If secondDash = 0 Then Exit Function

    ' RECIPE 取第二個連字號後的全部文字，例如 24W 或 SUB-A。
    GetSubCode = Trim$(Mid$(level1, secondDash + 1))
End Function

Private Function ImportFile(ByVal filePath As String, _
                            ByVal level1 As String, ByVal pn As String, _
                            ByVal level2 As String, ByVal lot As String) As String
    Dim raw As String, lines() As String, fields As Variant
    Dim sep As String, fileName As String, subCode As String
    Dim h As Long, i As Long, k As Long, lastField As Long
    Dim idCol As Long, xCol As Long, yCol As Long, binCol As Long
    Dim unitId As String, rowInFile As Long

    On Error GoTo FileFailed
    fileName = Mid$(filePath, InStrRev(Replace(filePath, "/", "\"), "\") + 1)
    subCode = GetSubCode(level1) ' 每個 CSV 只擷取一次。
    raw = ReadLocalText(filePath)
    If Len(raw) = 0 Then
        ImportFile = "空檔案"
        Exit Function
    End If

    raw = Replace(raw, vbCrLf, vbLf)
    lines = Split(Replace(raw, vbCr, vbLf), vbLf)
    raw = vbNullString

    h = -1
    For i = 0 To UBound(lines)
        If i >= 10 Then Exit For
        If InStr(1, lines(i), "2D ID", vbTextCompare) > 0 Then
            If InStr(lines(i), vbTab) > 0 Then
                sep = vbTab
            Else
                sep = ","
            End If
            fields = SplitRecord(lines(i), sep)
            idCol = -1: xCol = -1: yCol = -1: binCol = -1
            For k = 0 To UBound(fields)
                ' 這裡比對來源 CSV 的原始欄名，不使用輸出表頭。
                Select Case UCase$(FieldText(fields(k)))
                    Case "2D ID": idCol = k
                    Case "X": xCol = k
                    Case "Y": yCol = k
                    Case "UNIT BIN#": binCol = k
                End Select
            Next k
            If idCol >= 0 And xCol >= 0 And yCol >= 0 And binCol >= 0 Then
                h = i
                Exit For
            End If
        End If
    Next i

    If h < 0 Then
        ImportFile = "前 10 列找不到 2D ID、X、Y、Unit BIN#"
        Exit Function
    End If

    lastField = Application.Max(idCol, xCol, yCol, binCol)

    ' 明細解析或寫入出錯時中止整批，避免把部分匯入報為成功。
    On Error GoTo DetailFailed
    For i = h + 1 To UBound(lines)
        rowInFile = i + 1
        If Len(Trim$(lines(i))) > 0 Then
            fields = SplitRecord(lines(i), sep)
            If StrComp(FieldText(fields(0)), "Hard Limit", vbTextCompare) <> 0 Then
                If UBound(fields) >= idCol Then
                    unitId = FieldText(fields(idCol))
                    If Len(unitId) > 0 Then
                        If UBound(fields) < lastField Then
                            Err.Raise vbObjectError + 701,, "欄位不足。"
                        End If

                        AddRow level1, pn, level2, lot, subCode, fileName, unitId, _
                               FieldText(fields(xCol)), FieldText(fields(yCol)), _
                               FieldText(fields(binCol))
                    End If
                End If
            End If
        End If
    Next i
    Exit Function

FileFailed:
    ImportFile = Err.Description
    Exit Function

DetailFailed:
    Err.Raise Err.Number, "ImportFile", _
              "處理 CSV 第 " & rowInFile & " 列時出錯：" & Err.Description
End Function

Private Function ReadLocalText(ByVal filePath As String) As String
    Dim f As Integer, n As Long, opened As Boolean
    Dim bytes() As Byte, stream As Object
    Dim errNum As Long, errText As String

    On Error GoTo ReadFailed
    f = FreeFile
    Open filePath For Binary Access Read As #f
    opened = True
    n = LOF(f)
    If n > 0 Then
        ReDim bytes(0 To n - 1)
        Get #f,, bytes
    End If
    Close #f
    opened = False
    If n = 0 Then Exit Function

    ' 有 UTF-8 BOM 就以 UTF-8 解碼，否則沿用 Windows 系統編碼。
    If n >= 3 Then
        If bytes(0) = &HEF And bytes(1) = &HBB And bytes(2) = &HBF Then
            Set stream = CreateObject("ADODB.Stream")
            With stream
                .Type = 1
                .Open
                .Write bytes
                .Position = 0
                .Type = 2
                .Charset = "utf-8"
                ReadLocalText =.ReadText(-1)
                .Close
            End With
            If Left$(ReadLocalText, 1) = ChrW(&HFEFF) Then
                ReadLocalText = Mid$(ReadLocalText, 2)
            End If
            Exit Function
        End If
    End If

    ReadLocalText = StrConv(bytes, vbUnicode)
    Exit Function

ReadFailed:
    errNum = Err.Number
    errText = Err.Description
    On Error Resume Next
    If opened Then Close #f
    If Not stream Is Nothing Then stream.Close
    On Error GoTo 0
    Err.Raise errNum, "ReadLocalText", errText
End Function

Private Function SplitRecord(ByVal recordText As String, ByVal sep As String) As Variant
    Dim parts() As String, i As Long, first As Long, n As Long
    Dim ch As String, quoted As Boolean

    If InStr(recordText, """") = 0 Then
        SplitRecord = Split(recordText, sep)
        Exit Function
    End If

    ReDim parts(0 To 15)
    first = 1
    For i = 1 To Len(recordText)
        ch = Mid$(recordText, i, 1)
        If ch = """" Then
            quoted = Not quoted
        ElseIf ch = sep And Not quoted Then
            If n > UBound(parts) Then ReDim Preserve parts(0 To 2 * n - 1)
            parts(n) = Mid$(recordText, first, i - first)
            n = n + 1
            first = i + 1
        End If
    Next i

    If quoted Then
        Err.Raise vbObjectError + 702,, "有未閉合或跨列的引號欄位。"
    End If

    ReDim Preserve parts(0 To n)
    parts(n) = Mid$(recordText, first)
    SplitRecord = parts
End Function

Private Function FieldText(ByVal value As String) As String
    Dim s As String

    s = Trim$(value)
    If Left$(s, 2) = "=""" Then s = Mid$(s, 2) ' 例如 ="01"
    If Len(s) >= 2 Then
        If Left$(s, 1) = """" And Right$(s, 1) = """" Then
            s = Replace(Mid$(s, 2, Len(s) - 2), """""", """")
        End If
    End If
    FieldText = s
End Function

Private Sub AddRow(ByVal level1 As String, ByVal pn As String, _
                   ByVal level2 As String, ByVal lot As String, _
                   ByVal subCode As String, ByVal fileName As String, _
                   ByVal unitId As String, ByVal x As String, _
                   ByVal y As String, ByVal bin As String)

    If buffered = BATCH_SIZE Or nextRow + buffered > outWs.Rows.Count Then FlushBuffer
    If nextRow > outWs.Rows.Count Then
        pageNo = pageNo + 1
        PreparePage
    End If

    buffered = buffered + 1
    buf(buffered, 1) = level1 ' A: 第 1 層目錄
    buf(buffered, 2) = pn ' B: PN
    buf(buffered, 3) = level2 ' C: 第 2 層目錄
    buf(buffered, 4) = lot ' D: LOT
    If Len(subCode) > 0 Then
        buf(buffered, 5) = subCode ' E: RECIPE
    End If
    buf(buffered, 6) = fileName ' F: 檔案名稱
    buf(buffered, 7) = unitId ' G: 2D ID (MPID)
    buf(buffered, 8) = x ' H: X (MP_X)
    buf(buffered, 9) = y ' I: Y (MP_Y)
    buf(buffered, 10) = unitId & "_" & x & "_" & y ' J: MPID_X_Y

    ' K: Unit BIN#。保留原本的數字／文字處理方式。
    If Len(bin) > 0 And Len(bin) <= 9 And Not (bin Like "*[!0-9]*") Then
        buf(buffered, 11) = CLng(bin)
    ElseIf Len(bin) > 0 Then
        buf(buffered, 11) = "'" & bin
    End If
End Sub

Private Sub FlushBuffer()
    Dim tail() As Variant, r As Long, c As Long

    If buffered = 0 Then Exit Sub

    If buffered = BATCH_SIZE Then
        outWs.Cells(nextRow, 1).Resize(buffered, OUT_COLS).Value2 = buf
    Else
        ReDim tail(1 To buffered, 1 To OUT_COLS)
        For r = 1 To buffered
            For c = 1 To OUT_COLS
                tail(r, c) = buf(r, c)
            Next c
        Next r
        outWs.Cells(nextRow, 1).Resize(buffered, OUT_COLS).Value2 = tail
    End If

    nextRow = nextRow + buffered
    written = written + buffered
    buffered = 0
    ReDim buf(1 To BATCH_SIZE, 1 To OUT_COLS)
End Sub
