
Option Explicit

Private Const RBE_BATCH As Long = 20000
Private Const RBE_MAX_ROWS As Long = 1048576
Private Const RBE_MAX_RECIPES As Long = 16379
Private Const RBE_MAX_TEXT As Long = 32767
Private Const RBE_BLANK_HEADER As String = "ERROR_blank"
Private Const RBE_BLANK_KEY As Long = 0

Public Sub ExportPNLOTRecipeWithBlankWarning()
    Dim src As Workbook, dst As Workbook
    Dim ws As Worksheet, out As Worksheet
    Dim sources As Collection
    Dim details As Object, summaries As Object, recipes As Object
    Dim lots As Object, grp As Object, fs As Object
    Dim pnKey As Variant, lotKey As Variant, picked As Variant, a1 As Variant
    Dim savePath As String, stage As String, failure As String, msg As String
    Dim detailPages As Long, summaryPages As Long, nextRow As Long
    Dim blankRows As Long
    Dim oldCalc As XlCalculation, oldStatus As Variant
    Dim oldEvents As Boolean, oldScreen As Boolean, oldAlerts As Boolean
    Dim stateChanged As Boolean, saved As Boolean

    On Error GoTo Failed

    Set src = ThisWorkbook
    Set sources = New Collection

    For Each ws In src.Worksheets
        stage = "檢查來源表：" & ws.Name
        a1 = ws.Range("A1").Value2
        If Not IsError(a1) Then
            If StrComp(CStr(a1), "第 1 層目錄", vbBinaryCompare) = 0 Then
                RBE_CheckHeaders ws
                sources.Add ws
            End If
        End If
    Next ws

    stage = "辨識來源表"
    If sources.Count = 0 Then
        RBE_Fail "找不到 A1 為「第 1 層目錄」的來源表。"
    End If

    stage = "選擇儲存位置"
    picked = Application.GetSaveAsFilename( _
        InitialFileName:="PN_LOT_RECIPE_明細與彙整.xlsx", _
        FileFilter:="Excel Workbook (*.xlsx), *.xlsx")
    If VarType(picked) = vbBoolean Then Exit Sub

    savePath = CStr(picked)
    If LCase$(Right$(savePath, 5)) <> ".xlsx" Then savePath = savePath & ".xlsx"
    If InStr(1, savePath, "://", vbBinaryCompare) > 0 Then
        RBE_Fail "請選擇本機或已同步的資料夾。"
    End If

    Set fs = CreateObject("Scripting.FileSystemObject")
    If StrComp(savePath, src.FullName, vbTextCompare) = 0 Or _
       fs.FileExists(savePath) Or fs.FolderExists(savePath) Then
        RBE_Fail "目標名稱已存在，請選擇新檔名。"
    End If

    oldCalc = Application.Calculation
    oldStatus = Application.StatusBar
    oldEvents = Application.EnableEvents
    oldScreen = Application.ScreenUpdating
    oldAlerts = Application.DisplayAlerts
    stateChanged = True

    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    Application.DisplayAlerts = True

    Set details = RBE_Map()
    Set summaries = RBE_Map()
    Set recipes = RBE_Map()

    For Each ws In sources
        stage = "讀取來源表：" & ws.Name
        RBE_ReadSource ws, details, recipes, blankRows
    Next ws

    stage = "檢查來源資料"
    If details.Count = 0 Then RBE_Fail "沒有可匯出的資料。"

    ' 正式 RECIPE 全部讀完後才加入，確保異常欄固定在最右側。
    If blankRows > 0 Then
        If recipes.Count >= RBE_MAX_RECIPES Then
            RBE_Fail "加入 ERROR_blank 後，欄數超出 Excel 工作表上限。"
        End If
        recipes.Add RBE_BLANK_HEADER, recipes.Count + 1
    End If

    stage = "建立輸出活頁簿"
    Set dst = Workbooks.Add(xlWBATWorksheet)
    Set out = dst.Worksheets(1)
    RBE_StartPage dst, out, True, recipes, detailPages, nextRow, True

    For Each pnKey In details.Keys
        Set lots = details(pnKey)
        For Each lotKey In lots.Keys
            stage = "輸出中繼明細：" & CStr(pnKey) & " / " & CStr(lotKey)
            Application.StatusBar = stage
            Set grp = lots(lotKey)

            RBE_WriteGroup dst, out, True, recipes, detailPages, nextRow, _
                           CStr(pnKey), CStr(lotKey), grp

            stage = "彙整 RECIPE：" & CStr(pnKey) & " / " & CStr(lotKey)
            RBE_AddToSummary summaries, CStr(pnKey), CStr(lotKey), grp

            Set grp = Nothing
            lots.Remove lotKey
        Next lotKey
        Set lots = Nothing
        details.Remove pnKey
    Next pnKey
    RBE_FinishPage out, nextRow, 4 + recipes.Count

    stage = "建立 RECIPE 彙整表"
    RBE_StartPage dst, out, False, recipes, summaryPages, nextRow, False

    For Each pnKey In summaries.Keys
        Set lots = summaries(pnKey)
        For Each lotKey In lots.Keys
            stage = "輸出 RECIPE 彙整：" & CStr(pnKey) & " / " & CStr(lotKey)
            Application.StatusBar = stage
            Set grp = lots(lotKey)

            RBE_WriteGroup dst, out, False, recipes, summaryPages, nextRow, _
                           CStr(pnKey), CStr(lotKey), grp

            Set grp = Nothing
            lots.Remove lotKey
        Next lotKey
        Set lots = Nothing
        summaries.Remove pnKey
    Next pnKey
    RBE_FinishPage out, nextRow, 5 + recipes.Count

    stage = "儲存新檔"
    Application.StatusBar = stage
    If fs.FileExists(savePath) Or fs.FolderExists(savePath) Then
        RBE_Fail "目標名稱已存在，未覆蓋。"
    End If

    dst.Worksheets("RECIPE彙整_01").Activate
    dst.SaveAs Filename:=savePath, FileFormat:=xlOpenXMLWorkbook
    saved = True
    GoTo Finish

Failed:
    failure = stage & "：" & Err.Description

Finish:
    On Error Resume Next
    If Not saved Then
        If Not dst Is Nothing Then dst.Close SaveChanges:=False
    End If
    If stateChanged Then
        Application.Calculation = oldCalc
        Application.StatusBar = oldStatus
        Application.EnableEvents = oldEvents
        Application.ScreenUpdating = oldScreen
        Application.DisplayAlerts = oldAlerts
    End If
    On Error GoTo 0

    If saved Then
        msg = "已匯出中繼明細 " & detailPages & " 張、RECIPE 彙整 " & _
              summaryPages & " 張。" & vbCrLf & savePath
        If blankRows > 0 Then
            msg = msg & vbCrLf & vbCrLf & _
                  "有 " & Format$(blankRows, "#,##0") & _
                  " 筆來源列的 RECIPE 空白，已歸入最右側 ERROR_blank 欄。" & _
                  vbCrLf & "這些資料不參與 BinCode_sumary，請核對來源 E 欄。"
            MsgBox msg, vbExclamation
        Else
            MsgBox msg, vbInformation
        End If
    Else
        MsgBox "匯出未完成：" & failure, vbExclamation
    End If
End Sub

Private Sub RBE_CheckHeaders(ByVal ws As Worksheet)
    Dim refs As Variant, captions As Variant, v As Variant, i As Long

    refs = Array("B1", "C1", "E1", "J1", "K1")
    captions = Array("PN", "第 2 層目錄", "RECIPE", "MPID_X_Y", "Unit BIN#")

    For i = LBound(refs) To UBound(refs)
        v = ws.Range(CStr(refs(i))).Value2
        If IsError(v) Then RBE_Fail CStr(refs(i)) & " 包含 Excel 錯誤值。"
        If StrComp(CStr(v), CStr(captions(i)), vbBinaryCompare) <> 0 Then
            RBE_Fail CStr(refs(i)) & " 應為「" & CStr(captions(i)) & "」。"
        End If
    Next i
End Sub

Private Function RBE_LastSourceRow(ByVal ws As Worksheet) As Long
    Dim col As Variant, scan As Range, found As Range

    RBE_LastSourceRow = 1
    For Each col In Array("B", "C", "E", "J", "K")
        Set scan = ws.Columns(CStr(col))
        Set found = scan.Find(What:="*", After:=scan.Cells(1, 1), _
            LookIn:=xlFormulas, LookAt:=xlPart, SearchOrder:=xlByRows, _
            SearchDirection:=xlPrevious, MatchCase:=False, _
            MatchByte:=False, SearchFormat:=False)
        If Not found Is Nothing Then
            If found.Row > RBE_LastSourceRow Then RBE_LastSourceRow = found.Row
        End If
    Next col
End Function

Private Sub RBE_ReadSource(ByVal ws As Worksheet, ByVal details As Object, _
                           ByVal recipes As Object, ByRef blankRows As Long)
    Dim bc As Variant, e As Variant, jk As Variant, binValue As Variant
    Dim firstRow As Long, lastRow As Long, endRow As Long, n As Long
    Dim i As Long, rowNo As Long, itemNo As Long, recipeNo As Long
    Dim pn As String, fullLot As String, mpid As String, recipe As String
    Dim code As String, codes As String
    Dim lots As Object, grp As Object, ids As Object, bins As Object, hits As Object

    lastRow = RBE_LastSourceRow(ws)
    For firstRow = 2 To lastRow Step RBE_BATCH
        endRow = firstRow + RBE_BATCH - 1
        If endRow > lastRow Then endRow = lastRow
        n = endRow - firstRow + 1
        Application.StatusBar = "讀取 " & ws.Name & "：" & endRow & " / " & lastRow

        bc = ws.Cells(firstRow, 2).Resize(n, 2).Value2
        jk = ws.Cells(firstRow, 10).Resize(n, 2).Value2
        If n = 1 Then
            ReDim e(1 To 1, 1 To 1)
            e(1, 1) = ws.Cells(firstRow, 5).Value2
        Else
            e = ws.Cells(firstRow, 5).Resize(n, 1).Value2
        End If

        For i = 1 To n
            rowNo = firstRow + i - 1
            If IsError(bc(i, 1)) Or IsError(bc(i, 2)) Or _
               IsError(e(i, 1)) Or IsError(jk(i, 1)) Or IsError(jk(i, 2)) Then
                RBE_Fail "第 " & rowNo & " 列的必要欄位有 Excel 錯誤值。"
            End If

            pn = CStr(bc(i, 1))
            fullLot = CStr(bc(i, 2))
            recipe = Trim$(CStr(e(i, 1)))
            mpid = CStr(jk(i, 1))
            binValue = jk(i, 2)

            ' 五個輸入欄位都沒有資料時，不計為 RECIPE 空白異常。
            If Len(pn & fullLot & recipe & mpid & CStr(binValue)) = 0 Then
                GoTo NextRecord
            End If

            If Len(Trim$(pn)) = 0 Or Len(Trim$(fullLot)) = 0 Or _
               Len(Trim$(mpid)) = 0 Then
                RBE_Fail "第 " & rowNo & " 列的 PN、完整 LOT 或 MPID_X_Y 空白。"
            End If
            If IsEmpty(binValue) Or VarType(binValue) = vbBoolean Or _
               Not IsNumeric(binValue) Then
                RBE_Fail "第 " & rowNo & " 列的 Unit BIN# 空白或不是數值。"
            End If

            If Len(recipe) = 0 Then
                recipeNo = RBE_BLANK_KEY
                blankRows = blankRows + 1
            Else
                ' 保留名稱不得作為正式 RECIPE，避免與缺值欄混用。
                If StrComp(recipe, RBE_BLANK_HEADER, vbTextCompare) = 0 Then
                    RBE_Fail "E" & rowNo & " 使用了保留名稱 ERROR_blank。"
                End If
                If Not recipes.Exists(recipe) Then
                    If recipes.Count >= RBE_MAX_RECIPES Then
                        RBE_Fail "RECIPE 欄數超出 Excel 工作表上限。"
                    End If
                    recipes.Add recipe, recipes.Count + 1
                End If
                recipeNo = CLng(recipes(recipe))
            End If

            mpid = RBE_PadCoordinates(mpid)
            code = Trim$(Str$(CDbl(binValue)))

            Set lots = RBE_Child(details, pn)
            Set grp = RBE_Child(lots, fullLot)
            If Not grp.Exists("lot") Then
                If Len(fullLot) <= 1 Then
                    RBE_Fail "第 " & rowNo & " 列的第 2 層目錄太短，無法產生 LOT。"
                End If
                grp.Add "lot", Left$(fullLot, Len(fullLot) - 1)
            End If

            Set ids = RBE_Child(grp, "ids")
            Set bins = RBE_Child(grp, "bins")
            Set hits = RBE_Child(bins, recipeNo)
            itemNo = RBE_ItemNo(ids, mpid)

            If hits.Exists(itemNo) Then
                codes = CStr(hits(itemNo))
                If InStr(1, "," & codes & ",", "," & code & ",", vbBinaryCompare) = 0 Then
                    hits(itemNo) = RBE_JoinText(codes, code, "BIN 清單，第 " & rowNo & " 列")
                End If
            Else
                hits.Add itemNo, code
            End If
NextRecord:
        Next i
    Next firstRow
End Sub

Private Sub RBE_AddToSummary(ByVal summaries As Object, ByVal pn As String, _
                             ByVal fullLot As String, ByVal detailGroup As Object)
    Dim lots As Object, grp As Object
    Dim srcIds As Object, dstIds As Object, fullLists As Object, rowRanks As Object
    Dim srcBins As Object, dstBins As Object, srcHits As Object, dstHits As Object
    Dim ordered() As String, rowMap() As Long
    Dim i As Long, itemNo As Long, priority As Long
    Dim recipeKey As Variant, rowKey As Variant

    Set lots = RBE_Child(summaries, pn)
    Set grp = RBE_Child(lots, CStr(detailGroup("lot")))
    Set dstIds = RBE_Child(grp, "ids")
    Set dstBins = RBE_Child(grp, "bins")
    Set fullLists = RBE_Child(grp, "fullLists")
    Set rowRanks = RBE_Child(grp, "rowRanks")
    Set srcIds = detailGroup("ids")
    Set srcBins = detailGroup("bins")

    RBE_OrderedIds srcIds, ordered
    ReDim rowMap(1 To srcIds.Count)

    For i = 1 To srcIds.Count
        itemNo = RBE_ItemNo(dstIds, ordered(i))
        rowMap(i) = itemNo
        If fullLists.Exists(itemNo) Then
            fullLists(itemNo) = RBE_JoinText(CStr(fullLists(itemNo)), fullLot, _
                                "MPID_X_Y " & ordered(i) & " 的完整 LOT 清單")
        Else
            fullLists.Add itemNo, fullLot
        End If
    Next i

    For Each recipeKey In srcBins.Keys
        Set srcHits = srcBins(recipeKey)
        Set dstHits = RBE_Child(dstBins, recipeKey)
        For Each rowKey In srcHits.Keys
            itemNo = rowMap(CLng(rowKey))
            priority = RBE_BinRank(CStr(srcHits(rowKey)))
            If priority > 0 Then
                ' 缺值欄照樣彙整，但不影響整列正式判定。
                RBE_KeepHigherRank dstHits, itemNo, priority
                If CLng(recipeKey) <> RBE_BLANK_KEY Then
                    RBE_KeepHigherRank rowRanks, itemNo, priority
                End If
            End If
        Next rowKey
    Next recipeKey
End Sub

Private Function RBE_BinRank(ByVal codes As String) As Long
    Dim token As Variant

    If Len(codes) = 0 Then Exit Function
    For Each token In Split(codes, ",")
        Select Case Trim$(CStr(token))
            Case "32"
                RBE_BinRank = 3
                Exit Function
            Case "1"
                RBE_BinRank = 2
            Case ""
                ' 空白不列入判斷。
            Case Else
                If RBE_BinRank = 0 Then RBE_BinRank = 1
        End Select
    Next token
End Function

Private Sub RBE_KeepHigherRank(ByVal ranks As Object, ByVal itemNo As Long, _
                               ByVal priority As Long)
    If ranks.Exists(itemNo) Then
        If priority > CLng(ranks(itemNo)) Then ranks(itemNo) = priority
    Else
        ranks.Add itemNo, priority
    End If
End Sub

Private Function RBE_DisplayRank(ByVal priority As Long, _
                                 ByVal wholeRow As Boolean) As Variant
    Select Case priority
        Case 3: RBE_DisplayRank = 32
        Case 2: RBE_DisplayRank = 1
        Case 1
            If wholeRow Then
                RBE_DisplayRank = "Others"
            Else
                RBE_DisplayRank = "others"
            End If
        Case Else: RBE_DisplayRank = Empty
    End Select
End Function

Private Sub RBE_WriteGroup(ByVal dst As Workbook, ByRef ws As Worksheet, _
                           ByVal isDetail As Boolean, ByVal recipes As Object, _
                           ByRef pageNo As Long, ByRef nextRow As Long, _
                           ByVal pn As String, ByVal lotKey As String, _
                           ByVal grp As Object)
    Dim ids As Object, bins As Object, fullLists As Object, rowRanks As Object
    Dim ordered() As String, lot As String
    Dim n As Long, firstItem As Long, take As Long, room As Long
    Dim fixedCols As Long, blankRecipeCol As Long

    Set ids = grp("ids")
    Set bins = grp("bins")
    n = ids.Count
    RBE_OrderedIds ids, ordered

    fixedCols = RBE_FixedCols(isDetail)
    If recipes.Exists(RBE_BLANK_HEADER) Then
        blankRecipeCol = CLng(recipes(RBE_BLANK_HEADER))
    End If
    If isDetail Then
        lot = CStr(grp("lot"))
    Else
        lot = lotKey
        Set fullLists = grp("fullLists")
        Set rowRanks = grp("rowRanks")
    End If

    ' 能完整放入新頁的區塊，不在目前頁尾拆開。
    room = RBE_MAX_ROWS - nextRow + 1
    If n <= RBE_MAX_ROWS - 1 And n > room Then
        RBE_FinishPage ws, nextRow, fixedCols + recipes.Count
        RBE_StartPage dst, ws, isDetail, recipes, pageNo, nextRow, False
    End If

    firstItem = 1
    Do While firstItem <= n
        If nextRow > RBE_MAX_ROWS Then
            RBE_FinishPage ws, nextRow, fixedCols + recipes.Count
            RBE_StartPage dst, ws, isDetail, recipes, pageNo, nextRow, False
        End If

        room = RBE_MAX_ROWS - nextRow + 1
        take = n - firstItem + 1
        If take > room Then take = room

        RBE_WriteSegment ws, nextRow, isDetail, fixedCols, pn, lotKey, lot, _
                         ordered, bins, fullLists, rowRanks, firstItem, take, _
                         blankRecipeCol
        firstItem = firstItem + take
        nextRow = nextRow + take
    Loop
End Sub

Private Sub RBE_WriteSegment(ByVal ws As Worksheet, ByVal destRow As Long, _
                             ByVal isDetail As Boolean, ByVal fixedCols As Long, _
                             ByVal pn As String, ByVal lotKey As String, _
                             ByVal lot As String, ByRef ordered() As String, _
                             ByVal bins As Object, ByVal fullLists As Object, _
                             ByVal rowRanks As Object, ByVal firstItem As Long, _
                             ByVal itemCount As Long, ByVal blankRecipeCol As Long)
    Dim labels() As Variant, values() As Variant
    Dim hits As Object, recipeKey As Variant, v As Variant
    Dim lastItem As Long, lo As Long, hi As Long
    Dim n As Long, blockRow As Long, itemNo As Long, i As Long, recipeCol As Long
    Dim hasValues As Boolean

    lastItem = firstItem + itemCount - 1
    For lo = firstItem To lastItem Step RBE_BATCH
        hi = lo + RBE_BATCH - 1
        If hi > lastItem Then hi = lastItem

        n = hi - lo + 1
        blockRow = destRow + lo - firstItem
        ReDim labels(1 To n, 1 To fixedCols)

        For itemNo = lo To hi
            i = itemNo - lo + 1
            labels(i, 1) = "'" & pn
            If isDetail Then
                labels(i, 2) = "'" & lotKey
            Else
                labels(i, 2) = "'" & CStr(fullLists(itemNo))
                If rowRanks.Exists(itemNo) Then
                    labels(i, 5) = RBE_DisplayRank(CLng(rowRanks(itemNo)), True)
                End If
            End If
            labels(i, 3) = "'" & lot
            labels(i, 4) = "'" & ordered(itemNo)
        Next itemNo
        ws.Cells(blockRow, 1).Resize(n, fixedCols).Value2 = labels

        For Each recipeKey In bins.Keys
            Set hits = bins(recipeKey)
            recipeCol = CLng(recipeKey)
            If recipeCol = RBE_BLANK_KEY Then recipeCol = blankRecipeCol

            ReDim values(1 To n, 1 To 1)
            hasValues = False

            For itemNo = lo To hi
                If hits.Exists(itemNo) Then
                    v = hits(itemNo)
                    If isDetail Then
                        v = "'" & CStr(v)
                    Else
                        v = RBE_DisplayRank(CLng(v), False)
                    End If
                    values(itemNo - lo + 1, 1) = v
                    hasValues = True
                End If
            Next itemNo

            If hasValues Then
                ws.Cells(blockRow, fixedCols + recipeCol).Resize(n, 1).Value2 = values
            End If
        Next recipeKey
    Next lo
End Sub

Private Sub RBE_StartPage(ByVal dst As Workbook, ByRef ws As Worksheet, _
                          ByVal isDetail As Boolean, ByVal recipes As Object, _
                          ByRef pageNo As Long, ByRef nextRow As Long, _
                          ByVal reuseFirst As Boolean)
    Dim header() As Variant, key As Variant
    Dim fixedCols As Long, colCount As Long, prefix As String

    If Not reuseFirst Then
        Set ws = dst.Worksheets.Add(After:=dst.Worksheets(dst.Worksheets.Count))
    End If

    pageNo = pageNo + 1
    fixedCols = RBE_FixedCols(isDetail)
    If isDetail Then
        prefix = "中繼明細_"
    Else
        prefix = "RECIPE彙整_"
    End If
    ws.Name = prefix & Format$(pageNo, "00")

    colCount = fixedCols + recipes.Count
    ReDim header(1 To 1, 1 To colCount)
    header(1, 1) = "PN"
    header(1, 2) = "第 2 層目錄(LOT)"
    header(1, 3) = "LOT"
    header(1, 4) = "MPID_XY"
    If Not isDetail Then header(1, 5) = "BinCode_sumary"

    For Each key In recipes.Keys
        header(1, fixedCols + CLng(recipes(key))) = "'" & CStr(key)
    Next key

    With ws.Cells(1, 1).Resize(1, colCount)
        .Value2 = header
        .Interior.Color = RGB(31, 78, 120)
        .Font.Color = vbWhite
        .Font.Bold = True
        .WrapText = True
        .RowHeight = 30
    End With

    If recipes.Exists(RBE_BLANK_HEADER) Then
        With ws.Cells(1, fixedCols + CLng(recipes(RBE_BLANK_HEADER)))
            .Interior.Color = RGB(156, 74, 0)
            .Font.Color = vbWhite
        End With
    End If

    ws.Columns("A").ColumnWidth = 14
    ws.Columns("B").ColumnWidth = 24
    ws.Columns("C").ColumnWidth = 20
    ws.Columns("D").ColumnWidth = 30
    If Not isDetail Then
        ws.Columns("B").ColumnWidth = 46
        ws.Columns("E").ColumnWidth = 21
    End If
    ws.Cells(1, fixedCols + 1).Resize(1, recipes.Count).EntireColumn.ColumnWidth = 18
    nextRow = 2
End Sub

Private Sub RBE_FinishPage(ByVal ws As Worksheet, ByVal nextRow As Long, _
                           ByVal colCount As Long)
    ws.Range(ws.Cells(1, 1), ws.Cells(nextRow - 1, colCount)).AutoFilter
End Sub

Private Function RBE_FixedCols(ByVal isDetail As Boolean) As Long
    If isDetail Then
        RBE_FixedCols = 4
    Else
        RBE_FixedCols = 5
    End If
End Function

Private Function RBE_Map() As Object
    Dim d As Object
    Set d = CreateObject("Scripting.Dictionary")
    d.CompareMode = vbBinaryCompare
    Set RBE_Map = d
End Function

Private Function RBE_Child(ByVal parent As Object, ByVal key As Variant) As Object
    Dim child As Object

    If parent.Exists(key) Then
        Set child = parent(key)
    Else
        Set child = RBE_Map()
        parent.Add key, child
    End If
    Set RBE_Child = child
End Function

Private Function RBE_ItemNo(ByVal ids As Object, ByVal mpid As String) As Long
    If Not ids.Exists(mpid) Then ids.Add mpid, ids.Count + 1
    RBE_ItemNo = CLng(ids(mpid))
End Function

Private Sub RBE_OrderedIds(ByVal ids As Object, ByRef ordered() As String)
    Dim key As Variant

    ReDim ordered(1 To ids.Count)
    For Each key In ids.Keys
        ordered(CLng(ids(key))) = CStr(key)
    Next key
End Sub

Private Function RBE_PadCoordinates(ByVal mpid As String) As String
    Dim parts As Variant, n As Long

    parts = Split(mpid, "_")
    n = UBound(parts)
    If n >= 2 Then
        If parts(n - 1) Like "#" Then parts(n - 1) = "0" & parts(n - 1)
        If parts(n) Like "#" Then parts(n) = "0" & parts(n)
    End If
    RBE_PadCoordinates = Join(parts, "_")
End Function

Private Function RBE_JoinText(ByVal existingText As String, _
                              ByVal nextText As String, _
                              ByVal fieldLabel As String) As String
    If Len(existingText) + 1 + Len(nextText) > RBE_MAX_TEXT Then
        RBE_Fail fieldLabel & "超出 Excel 單格字數上限。"
    End If
    RBE_JoinText = existingText & "," & nextText
End Function

Private Sub RBE_Fail(ByVal message As String)
    Err.Raise vbObjectError + 831, "ExportPNLOTRecipeWithBlankWarning", message
End Sub
